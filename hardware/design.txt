
Interface:
    Code Interface:
        - input   CODE.DATA[6:0]
        - input   CODE.STROBE      (* pulse input *)
        - output  CODE.LABEL[6:0]
        - output  CODE.SEEK        (* pulse output *)
        - output  CODE.HALT        (* pulse output *)

    Memory Interface:
        - output MEM.ADDR[7:0]
        - inout  MEM.DATA[7:0]
        - output MEM.CS
        - output MEM.OE
        - output MEM.WE

High-Level Components:
    IR[5:0]:  6 bit, Instruction Register
    OP[7:0]:  8 bit, Opcode Register
    ACC[7:0]: 8 bit, Accumulator Register
    ALU:      8 bit ROM with 19 bit addressing.

Data Movements + State Machine:
    state[fetch instruction]:
        when[CODE.STROBE]:
            IR <= CODE.DATA[5:0]                 (* truncate MSB *)
            goto state[fetch operand]
    state[fetch operand]:
        when[CODE.STROBE]:
            OP <= { IR[5], CODE.DATA }
            goto IR[4] ? state[indirect operand] : state[execute]
    state[indirect operand]:
        MEM.CS   <= 1
        MEM.OE   <= 1
        MEM.WE   <= 0
        MEM.ADDR <= OP
        OP       <= MEM.DATA
        goto state[execute]
    state[execute]:
        if IR[3]:
            ALU.ADDR <= {IR[2:0], ACC, OP}
            ALU.OE   <= 1
            ACC      <= ALU.DATA
        elif IR[2:0] == 0: (* HALT *)
            CODE.HALT <= 1 (* emit pulse *)
        elif IR[2:0] == 1: (* SET *)
            ACC      <= OP
        elif IR[2:0] == 2: (* LD *)
            MEM.CS   <= 1
            MEM.OE   <= 1
            MEM.WE   <= 0
            MEM.ADDR <= OP
            ACC      <= MEM.DATA
        elif IR[2:0] == 3: (* ST *)
            MEM.CS   <= 1
            MEM.OE   <= 0
            MEM.WE   <= 1
            MEM.ADDR <= OP
            MEM.DATA <= ACC
        elif IR[2]:
            CODE.LABEL <= OP[6:0];               (* truncate MSB *)
            if IR[1:0] == 0: (* JMP *)
                CODE.SEEK <= 1                   (* always emit pulse *)
            elif IR[1:0] == 1: (* JZ *)
                CODE.SEEK <= (ACC == 0)          (* emit pulse when zero *)
            elif IR[1:0] == 2: (* JNZ  *)
                CODE.SEEK <= (ACC != 0)          (* emit pulse when not zero *)
            else: (* DJNZ *)
                CODE.SEEK <= (ACC != 1)          (* emit pulse when going to zero *)
                ALU.ADDR  <= { 3'b1, ACC, 8'b1 } (* subtract one *)
                ALU.OE    <= 1
                ACC       <= ALU.DATA
        goto state[fetch instruction]

Logic Formules and State Sequencing:
    This is the logic without *signal sequencing*, so no care for timing still.

    Add Components:
        STATE_FI: D-FlipFlop, Fetch Instruction
        STATE_FO: D-FlipFlop, Fetch Operand 
        STATE_IO: D-FlipFlop, Indirect Operand
        STATE_EX: D-FlipFlop, Execute

        OP_BUF:   3-State-Buffer, Operand
        ACC_BUF:  3-State-Buffer, Accumulator

    Logic:
        ADVANCE_STATE <= (* synthetic advance pulse generated by monoflops *)
        CLK           <= CODE.STROBE | ADVANCE_STATE

        (* aliases for instructions *)
        OPCODE_ALU    <= IR[3]
        OPCODE_HALT   <= !IR[3] & (IR[2:0] == 0)            (* provided by 1:8 demux *)
        OPCODE_SET    <= !IR[3] & (IR[2:0] == 1)            (* provided by 1:8 demux *)
        OPCODE_LD     <= !IR[3] & (IR[2:0] == 2)            (* provided by 1:8 demux *)
        OPCODE_ST     <= !IR[3] & (IR[2:0] == 3)            (* provided by 1:8 demux *)
        OPCODE_JMP    <= !IR[3] & (IR[2:0] == 4)            (* provided by 1:8 demux *)
        OPCODE_JZ     <= !IR[3] & (IR[2:0] == 5)            (* provided by 1:8 demux *)
        OPCODE_JNZ    <= !IR[3] & (IR[2:0] == 6)            (* provided by 1:8 demux *)
        OPCODE_DJNZ   <= !IR[3] & (IR[2:0] == 7)            (* provided by 1:8 demux *)

        STATE_FI.D    <= STATE_EX.Q
        STATE_FI.CLK  <= CLK

        STATE_FO.D    <= STATE_FI.Q
        STATE_FO.CLK  <= CLK

        STATE_IO.D    <= IR[4] ? STATE_FO.Q : 0             (* mux 2:1 *)
        STATE_IO.CLK  <= CLK

        STATE_EX.D    <= IR[4] ? STATE_IO.Q : STATE_FO.Q    (* mux 2:1 *)
        STATE_EX.CLK  <= CLK

        BUS           <= ALU.DATA ~ MEM.DATA ~ OP_BUF.Q ~ ACC_BUF.Q     (* tristate join *)

        IR.D          <= CODE.DATA[5:0]
        IR.EN         <= STATE_FI.Q
        IR.CLK        <= STATE_FI.Q & CODE.STROBE

        OP.D          <= STATE_FO.Q ? { IR[5], CODE.DATA } : MEM.DATA
        OP.EN         <= STATE_FO.Q | STATE_IO.Q
        OP.CLK        <= CLK

        OP_BUF.D      <= OP.Q
        OP_BUF.OE     <= STATE_EX.Q & OPCODE_SET

        ACC.D         <= BUS
        ACC.EN        <= STATE_EX.Q & (OPCODE_ALU | OPCODE_SET | OPCODE_LD | OPCODE_DJNZ)
        ACC.CLK       <= CLK

        ACC_BUF.D     <= ACC.Q
        ACC_BUF.OE    <= STATE_EX.Q & OPCODE_ST

        MEM.CS        <= STATE_IO.Q | (STATE_EX.Q & (OPCODE_LD | OPCODE_ST))
        MEM.OE        <= STATE_IO.Q | OPCODE_LD
        MEM.WE        <= STATE_EX.Q & OPCODE_ST
        MEM.ADDR      <= OP.Q

        ALU.ADDR      <= OPCODE_ALU ? {IR[2:0], ACC, OP} : {3'b1, ACC, 8'b1} (* mux 2:1 *)
        ALU.OE        <= STATE_EX.Q & (OPCODE_ALU | OPCODE_DJNZ)

        CODE.LABEL    <= OP[6:0]
        CODE.SEEK     <= STATE_EX.Q & (OPCODE_JMP | (OPCODE_JZ & (ACC==0)) | (OPCODE_JNZ & !(ACC==0)) | (OPCODE_DJNZ & !(ACC==1)))
        CODE.HALT     <= STATE_EX.Q & OPCODE_HALT
